## 1. Create VLAN 
--------------------------------------
Switch>en
Switch#config t
Enter configuration commands, one per line.  End with CNTL/Z.
Switch(config)#vlan 60
Switch(config-vlan)#name IT/Operations

--------------------------------------

Switch(config-if-range)#switchport mode access
Switch(config-if-range)#switchport access vlan 60

## 5. Set up OSPF
-----------------------
SW3>en
SW3#config t
Enter configuration commands, one per line.  End with CNTL/Z.
SW3(config)#int g0/1
SW3(config-if)#no swi
SW3(config-if)#no switchport 
SW3(config-if)#
%LINEPROTO-5-UPDOWN: Line protocol on Interface GigabitEthernet0/1, changed state to down

%LINEPROTO-5-UPDOWN: Line protocol on Interface GigabitEthernet0/1, changed state to up

SW3(config-if)#ip ad
SW3(config-if)#ip address 192.168.100.10 255.255.255.252
SW3(config-if)#no shutdown
SW3(config-if)#int g0/2
SW3(config-if)#no switchport 
SW3(config-if)#
%LINEPROTO-5-UPDOWN: Line protocol on Interface GigabitEthernet0/2, changed state to down

%LINEPROTO-5-UPDOWN: Line protocol on Interface GigabitEthernet0/2, changed state to up

SW3(config-if)#ip address 192.168.100.2 255.255.255.252
SW3(config-if)#no shutdown
SW3(config-if)#

SW3(config-if)#router ospf 10
SW3(config-router)# router-id 3.3.3.3
SW3(config-router)# network 192.168.60.0 0.0.0.255 area 0
SW3(config-router)# network 192.168.100.0 0.0.0.3 area 0
SW3(config-router)# network 192.168.100.8 0.0.0.3 area 0
SW3(config-router)# passive-interface vlan 60
SW3(config-router)#
